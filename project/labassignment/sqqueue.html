头文件round-robin_queue.h 编码如下:

#include <iostream>
#define elemtype int
#define maxsize 100
using namespace std;

typedef struct roundqueue {
	int front;
	int rear;
	elemtype* base;
	int queuesize;
};

void initqueue(roundqueue& S) {//初始化
	S.base = new elemtype[maxsize];
	if (!S.base)return;
	S.queuesize = maxsize;
	S.front = S.rear = 0;
}

void enqueue(roundqueue& S, elemtype e) {//插入元素e
	if ((S.rear + 1) % S.queuesize == S.front)
	{
		return;
	}
	S.base[S.rear] = e;
	S.rear = (S.rear + 1) % S.queuesize;
}

void dequeue(roundqueue& S, elemtype& e) {//删除元素
	if (S.rear == S.front)
	{
		return;
	}
	e = S.base[S.front];
	S.front = (S.front + 1) % S.queuesize;
}

elemtype getelem(roundqueue S) {//取出队首元素
	return(S.base[S.front]);
}

int queuelen(roundqueue S) {//求队列长
	return((S.rear - S.front + S.queuesize) % S.queuesize);
}

void destroyqueue(roundqueue& S) {//销毁队列
	delete[]S.base;
	S.base = nullptr;
	S.front = S.rear = 0;
	S.queuesize = 0;
}

// 遍历打印，不破坏队列，过滤0
void travelqueue(roundqueue& S, int lineWidth) {
	int len = queuelen(S);
	elemtype e;
	for (int k = 0; k < lineWidth; k++) {
		cout << "  ";
	}
	for (int i = 0; i < len; i++) {
		dequeue(S, e);
		if (e != 0) {
			printf("%4d", e);
		}
		enqueue(S, e);
	}
	cout << endl;
}
// 根据当前队列，生成下一行
void pascal_tri(roundqueue& S) {
	elemtype e = 0; elemtype r;
	int len = queuelen(S);
	for (int i = 0; i < len; i++) {
		dequeue(S, r); enqueue(S, e + r);
		e = r;
	}
	enqueue(S, 0);
}

杨辉三角计算输出操作
#include"round-robin_queue.h"

int main() {
	roundqueue S;
	initqueue(S);
	int n;
	cout << "请输入行数：";
	cin >> n;
	cout << '\n';
	enqueue(S, 0);
	enqueue(S, 1);
	enqueue(S, 0);
	for (int i = 0; i < n; i++) {
		travelqueue(S,n-i);
		pascal_tri(S);
	}
	destroyqueue(S);
	return 0;
}

elemtype getelem(roundqueue S) O (1) 获取队首
int queuelen(roundqueue S) O ( 1 ) 求队列长度
