1､ 基本操作实现函数代码:

头文件sqstack.h 编码如下:

#include <iostream>
using namespace std;
#define maxsize 100
#define maxstacksize 10
#define elemtype int

typedef struct {
	elemtype* base;
	elemtype* top;
	int stacksize;
}sqstack;

void initstack(sqstack& S) {//初始化一个顺序栈
	S.base = new elemtype[maxstacksize];
	if (!S.base)return;
	S.top = S.base;
	S.stacksize = maxstacksize;
}

void gettop(sqstack S,elemtype& e) {//取栈顶元素，用e返回
	if (S.base == S.top)return;
	e = *(S.top - 1);
}

void push(sqstack& S, elemtype e) {//将元素e放入栈顶
	if (S.top - S.base >= S.stacksize) {
		int newsize = S.stacksize + maxstacksize;
		elemtype* a = new elemtype[newsize];
		int len = S.top - S.base;
		for (int i = 0; i < len; i++) a[i] = S.base[i];
		delete[]S.base;
		S.base = a; S.top =S.base+len;
		S.stacksize=newsize;
	}
	*S.top++ = e;
}

void pop(sqstack& S, elemtype& e) {//删除栈顶元素，并用e返回
	if (S.base == S.top)return;
	e = *(S.top - 1);
	S.top--;
}

void clear(sqstack& S) {//清空栈
	elemtype e;
	if (S.top == S.base)return;
	pop(S, e);
	clear(S);
}

int stacklen(sqstack S) {//得出栈的长度
	return(S.top - S.base);
}

bool stackempty(sqstack S) {//判断栈是否为空，空为真
	if ((S.top - S.base)==0)return true;
	else return false;
}
void destory(sqstack& S) {//销毁栈
	if (S.base!=nullptr) {
		delete[]S.base;
		S.base = nullptr;
		S.top = nullptr;
		S.stacksize = 0;
	}
}

void traver_print(sqstack S) {//遍历并输出整个栈
	elemtype a; int l = stacklen(S);
	for (int i = 0; i < l; i++) {
		pop(S, a);
		cout << a;
	}
}

void transform(int num) {//十进制转化为八进制（元素必须是0~9的整数）
	if (!num)return;
	sqstack Q;
	initstack(Q);
	int p;
	while (num>0) {
		push(Q, num % 8);
		num = num / 8;
	}
	traver_print(Q);
	destory(Q);	
}
