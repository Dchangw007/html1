#include 
#include    // exit
#define LIST_INIT_SIZE 100    //初始分配量
#define LISTINCREMENT 10      // 分配增量
#define OK 1
#define ERROR 0
#define OVERFLOW -1
typedef int ElemType;
typedef int Status;
typedef struct {
	ElemType* elem;    // 存储空间基址
	int  length;        // 当前长度
	int  listsize;      // 当前分配的存储容量(以sizeof(ElemType)为单位)
}List;

Status InitList(List& L){// 构造一个空的顺序表
	// C++ new 代替 malloc
	L.elem = new ElemType[LIST_INIT_SIZE];
	if (!L.elem) exit(OVERFLOW);
	L.length = 0;
	L.listsize = LIST_INIT_SIZE;
	return OK;
}


void DestroyList(List& L){// 销毁顺序表
	if (L.elem)
	{
		delete[] L.elem;   // C++ delete[] 代替 free
		L.elem = nullptr; // 置空，防止野指针
		L.length = 0;
		L.listsize = 0;
	}
}


int ListLength(List L){// 得到顺序表L的元素个数
	return L.length;
}


Status GetElem(List L, int i, ElemType& e){// 取出顺序表L的第 i 个元素，赋值给元素e, i 的合法范围为 1≤i≤L.length
	if (i < 1 || i > L.length)
		return ERROR;
	e = L.elem[i - 1];
	return OK;
}


Status ListInsert(List& L, int i, ElemType e){// 在顺序表L的第 i 个元素之前插入新的元素e
	ElemType* p, * q;
	if (i < 1 || i > L.length + 1)
		return ERROR;
	if (L.length >= L.listsize)
		return OVERFLOW;   // 存储空间已满
	q = &(L.elem[i - 1]);                 // q 指示插入位置
	for (p = &(L.elem[L.length - 1]); p >= q; --p)
		*(p + 1) = *p;       // 元素右移
	*q = e;       // 插入e
	++L.length;   // 表长增1
	return OK;
}

int LocateElem(List L, ElemType e){// 在顺序表中查询第一个满足判定条件的数据元素
	int i = 1;
	ElemType* p = L.elem;
	while (i <= L.length && !(*p++==e))
		++i;
	if (i <= L.length)
		return i;
	else
		return 0;
}

void ListTraverse(List L){// 遍历输出顺序表
	ElemType* p;
	p = L.elem;
	printf("List: ");
	for (int i = 0; i < L.length; i++, p++)
		printf("%d   ", *p);
	printf("\n");
}
Status deletelisti(List& L, int i)
{
	ElemType* p, * q;
	q = &(L.elem[i - 1]);                 // q 指示删除位置
	for (p = &(L.elem[L.length - 1]); p >= q; ++q,++i) {
		L.elem[i] = L.elem[i + 1];
	}
	--L.length;   // 表长减1
	return OK;
}Status deleteliste(List& L, ElemType e) {
	deletelisti(L, LocateElem(L,e)-1);
	return OK;
}

2､ 操作复杂度总结

在此处进行时间复杂度分析,设线性表长度为 n ,各操作复杂度见下表

操作(函数名) 时间复杂度 说明
Status InitList(List& L) O (1) 初始化空顺序表
void DestroyList(List& L) O (1) 销毁顺序表
int ListLength(List L) O (1) 获取顺序表长度
Status GetElem(List L, int i, ElemType& e) O (1) 获取第 i 个元素的值
Status ListInsert(List& L, int i, ElemType e) O ( n ) 在顺序表第 i 个位置之前插入元素 e
int LocateElem(List L, ElemType e) O ( n ) 查找值等于 e 的第一个元素，返回它的位序
void ListTraverse(List L) O ( n ) 遍历整个顺序表，打印全部元素。
Status deletelisti(List& L, int i) O ( n ) 删除顺序表第 i 个位置的元素
Status deleteliste(List& L, ElemType e) O ( n ) 删除第一个值等于 e 的元素
