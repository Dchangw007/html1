#include <stdio.h>
#include <stdlib.h>
#include <iostream>
using namespace std;
#define maxsize 100
#define initstacksize 10
typedef struct elemtype {//多项式项：系数、指数
	int expn;
	float coef;
};
typedef struct Lnode {//链表节点
	elemtype elem;
	struct Lnode* next;
}*linklist, * position;
typedef struct {//带头尾指针+长度的链表结构
	linklist head, tail;
	int len;
}link;
void initelemtype(elemtype a[], int n) {//初始化元素数组：n为项数
	for (int i = 0; i < n; i++) {
		cin >> a[i].coef >> a[i].expn;
	}
}
void initlist(link* L) {//初始化链表：带头结点
	L->head = L->tail = new Lnode;
	if (!L->head) return;
	L->head->next = NULL;
	L->len = 0;
}
void destroylist(link* L) {//销毁链表
	linklist p = L->head;
	while (p) {
		linklist q = p->next;
		delete p;
		p = q;
	}
	L->head = L->tail = NULL;
	L->len = 0;
}
void insertlist(link* L, int i, elemtype& e) {//在第i个位置前插入元素e（i从1开始）
	if (i<1 || i>L->len + 1) return;
	linklist p = L->head;
	for (int k = 1; k < i; k++) {
		p = p->next;
	}
	linklist s = new Lnode;
	if (!s) return;
	s->elem = e;
	s->next = p->next;
	p->next = s;
	if (p == L->tail) {
		L->tail = s;
	}
	L->len += 1;
}
void deletelist(link* L, int i, elemtype& e) {//删除第i个元素，值存入e
	if (i<1 || i>L->len) return;
	linklist p = L->head;
	for (int k = 1; k < i; k++) {
		p = p->next;
	}
	linklist q = p->next;
	e = q->elem;
	p->next = q->next;
	if (q == L->tail) {
		L->tail = p;
	}
	delete q;
	L->len--;
}
int locateelem(link* L, int e) {//查找第一个指数等于e的项，返回位序，找不到返回0
	linklist p = L->head->next;
	int i = 1;
	while (p) {
		if (p->elem.expn == e)
			return i;
		p = p->next;
		i++;
	}
	return 0;
}
void traverselist(link L) {//遍历输出多项式
	linklist p = L.head->next;
	while (p) {
		cout << p->elem.coef << "x^" << p->elem.expn << " ";
		p = p->next;
	}
	cout << endl;
}
void ordered(elemtype a[], int n) {//冒泡排序：按指数expn升序，n是数组长度
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < n - i - 1; j++) {
			if (a[j].expn > a[j + 1].expn) {
				elemtype t = a[j];
				a[j] = a[j + 1];
				a[j + 1] = t;
			}
		}
	}
}
void addlist(link& Lc, link La, link Lb) {//多项式加法 La + Lb = Lc，La、Lb已经按指数升序
	initlist(&Lc);
	linklist pa = La.head->next;
	linklist pb = Lb.head->next;
	while (pa != NULL && pb != NULL) {
		if (pa->elem.expn < pb->elem.expn) {//pa指数小，插入Lc
			insertlist(&Lc, Lc.len + 1, pa->elem);
			pa = pa->next;
		}
		else if (pa->elem.expn > pb->elem.expn) {//pb指数小，插入Lc
			insertlist(&Lc, Lc.len + 1, pb->elem);
			pb = pb->next;
		}
		else {//指数相等，系数相加
			elemtype sum = pa->elem;
			sum.coef += pb->elem.coef;
			if (sum.coef != 0) { //系数不为0才保留该项
				insertlist(&Lc, Lc.len + 1, sum);
			}
			pa = pa->next;
			pb = pb->next;
		}
	}
	while (pa != NULL) {//把剩下的节点直接追加
		insertlist(&Lc, Lc.len + 1, pa->elem);
		pa = pa->next;
	}
	while (pb != NULL) {
		insertlist(&Lc, Lc.len + 1, pb->elem);
		pb = pb->next;
	}
}
void multilist(link& Lmul, link La, link Lb)
{
	initlist(&Lmul);
	linklist pa = La.head->next;

	while (pa != NULL){//外层循环遍历A每一项
		linklist pb = Lb.head->next;
		while (pb != NULL){//内层循环遍历B每一项
			//两项相乘：系数相乘，指数相加
			elemtype temp;
			temp.coef = pa->elem.coef * pb->elem.coef;
			temp.expn = pa->elem.expn + pb->elem.expn;
			//查找乘积链表中是否已有相同指数
			int pos = locateelem(&Lmul, temp.expn);
			if (pos != 0){//找到同指数，取出该项，累加系数
				elemtype e;
				deletelist(&Lmul, pos, e);
				e.coef += temp.coef;
				if (e.coef != 0){
					insertlist(&Lmul, pos, e);
				}
			}
			else{//没有同指数，直接插入到末尾
				insertlist(&Lmul, Lmul.len + 1, temp);
			}
			pb = pb->next;
		}
		pa = pa->next;
	}
}

一元多项式线性操作
#include"sqstack.h"
int main(){
	link La, Lb, Lc, Lmul;
	int na, nb;
	cout << "请输入多项式A的项数：";
	cin >> na;
	elemtype a[maxsize];
	cout << "依次输入每一项 系数 指数：" << endl;
	initelemtype(a, na);
	ordered(a, na);
	initlist(&La);
	for (int i = 0; i < na; i++) {
		insertlist(&La, La.len + 1, a[i]);
	}
	cout << "请输入多项式B的项数：";
	cin >> nb;
	elemtype b[maxsize];
	cout << "依次输入每一项 系数 指数：" << endl;
	initelemtype(b, nb);
	ordered(b, nb);
	initlist(&Lb);
	for (int i = 0; i < nb; i++) {
		insertlist(&Lb, Lb.len + 1, b[i]);
	}
	cout << "多项式A："; traverselist(La);
	cout << "多项式B："; traverselist(Lb);
	addlist(Lc, La, Lb);
	cout << "A+B结果："; traverselist(Lc);
	multilist(Lmul, La, Lb);
	cout << "A*B结果："; traverselist(Lmul);
	destroylist(&La);
	destroylist(&Lb);
	destroylist(&Lc);
	destroylist(&Lmul);
	return 0;
}

2､ 操作复杂度总结

在此处进行时间复杂度分析,设线性表长度为 n ,各操作复杂度见下表

操作(函数名) 时间复杂度 说明
initelemtype(elemtype a[], int n) O (n) 输入 n 组系数、指数存入数组
initlist(link* L) O (1) 创建头结点，初始化头尾指针与长度
destroylist(link* L) O (n) 逐个释放链表所有节点
insertlist(link* L, int i, elemtype& e) O (n) 第 i 位置前插入元素e
deletelist(link* L, int i, elemtype& e) O ( n ) 删除第 i 个元素
locateelem(link* L, int e) O ( n ) 查找指数等于 e 的项
traverselist(link L) O ( n ) 遍历输出所有链表元素
ordered(elemtype a[], int n) O ( n² ) 冒泡排序
addlist(link& Lc, link La, link Lb) O ( 4n² ) 多项式相加
multilist(link& Lmul, link La, link Lb) O ( n² ) 多项式相乘
